1""" 

2Where legacy rows come from: the legacy MySQL database, or (in tests) plain Python lists. 

3 

4Both give the importer the same thing: for a table name, an iterable of rows as dicts keyed by column name. 

5""" 

6 

7import environ 

8 

9 

10class ListSource: 

11 """Rows held in memory: {"users": [{...}, ...], ...}. Used by the tests.""" 

12 

13 def __init__(self, tables: dict): 

14 self.tables = tables 

15 

16 def rows(self, table: str, order_by: str = "id"): 

17 return iter(sorted(self.tables.get(table, []), key=lambda row: row.get(order_by) or 0)) 

18 

19 def count(self, table: str) -> int: 

20 return len(self.tables.get(table, []))

23class MySQLSource: 

24 """The legacy database, read-only. `url` looks like mysql://user:password@127.0.0.1:3306/bbms.""" 

25 

26 def __init__(self, url: str): 

27 import pymysql # only needed when importing, so imported here 

28 

29 config = environ.Env.db_url_config(url) 

30 self._pymysql = pymysql 

31 self.connection = pymysql.connect( 

32 host=config.get("HOST") or "127.0.0.1", 

33 port=int(config.get("PORT") or 3306), 

34 user=config["USER"], 

35 password=config.get("PASSWORD") or "", 

36 database=config["NAME"], 

37 charset="utf8mb4", 

38 ) 

39 with self.connection.cursor() as cursor: 

40 # Never write to the legacy database, even if the user given is allowed to (use a read-only user anyway). 

41 cursor.execute("SET SESSION TRANSACTION READ ONLY") 

42 # Times are stored in UTC (the legacy app's timezone); keep them as naive values and let the importer 

43 # attach UTC. Zero dates ('0000-00-00') come back as strings, which the importer treats as empty. 

44 cursor.execute("SET time_zone = '+00:00'") 

45 

46 def rows(self, table: str, order_by: str = "id"): 

47 # A server-side cursor streams big tables (the access log) instead of loading them all at once. 

48 with self.connection.cursor(self._pymysql.cursors.SSDictCursor) as cursor: 

49 cursor.execute(f"SELECT * FROM `{table}` ORDER BY `{order_by}`") 

50 yield from cursor 

51 

52 def count(self, table: str) -> int: 

53 with self.connection.cursor() as cursor: 

54 cursor.execute(f"SELECT COUNT(*) FROM `{table}`") 

55 return cursor.fetchone()[0]
