1""" 

2Values the trustees can change without a code change (spec 02 §2.8). 

3 

4One row (`SiteSettings.load()`) with one typed field per setting, so each value is validated and has a label and 

5help text in the admin. Every change is recorded in `SettingChange` (who, when, old and new value). 

6 

7To add a setting: add a field here with the current value as its default, add it to a group in GROUPS, make a 

8migration, and read it with `SiteSettings.load().<field>` where the constant used to be. Money fields end in 

9`_pence` (the admin shows them in pounds); add settings only when something uses them. 

10 

11Settings are read from the database each time (one quick query), so a change applies at once, on every server. 

12"""

358class SettingChange(models.Model): 

359 """The audit log for settings (02 §2.8: changes are audit-logged).""" 

360 

361 setting = models.CharField(max_length=100) 

362 old_value = models.CharField(max_length=255) 

363 new_value = models.CharField(max_length=255) 

364 changed_by = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.SET_NULL, null=True, related_name="+") 

365 changed_at = models.DateTimeField(default=timezone.now) 

366 

367 class Meta: 

368 ordering = ["-changed_at"] 

369 

370 def __str__(self): 

371 return f"{self.setting}: {self.old_value} → {self.new_value}"
