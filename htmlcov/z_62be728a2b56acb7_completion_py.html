1""" 

2Course completion (spec 06 §6.12, "Checking completion"). 

3 

4BBMS asks Moodle, and **stores** what it learns as a `TrainingCompletion`. Machine access and sign-off use only the 

5stored record (`equipment.rules`), never a live call, so a Moodle outage can't block a machine. Once recorded, a 

6completion stays until an admin clears it; a cleared one only comes back if Moodle reports the course finished again 

7after it was cleared. 

8 

9BBMS checks: when a member presses "check now", when a trainer opens the sign-off list, and every 15 minutes for the 

10members waiting for sign-off (`manage.py check_moodle_completions`). 

11"""

41def check(member, course_id, now=None) -> str: 

42 """Ask Moodle whether this member has finished this course, and record it if so.""" 

43 now = now or timezone.now() 

44 existing = TrainingCompletion.objects.filter(member=member, moodle_course_id=course_id).first() 

45 if existing and existing.counts: 

46 return ALREADY 

47 api = moodle.client() 

48 try: 

49 account = MoodleAccount.objects.filter(member=member).first() 

50 if account is None: 

51 user_id = api.find_user_id(member) 

52 if user_id is None: 

53 return NO_ACCOUNT 

54 account = MoodleAccount.objects.create(member=member, moodle_user_id=user_id, found_at=now) 

55 status = api.course_status(account.moodle_user_id, course_id) 

56 except moodle.MoodleError as error: 

57 logger.warning("Checking course %s for member %s failed: %s", course_id, member.pk, error) 

58 return UNAVAILABLE 

59 if not status.completed: 

60 return NOT_YET 

61 finished = status.completed_at or now 

62 if existing is not None: # cleared earlier: only a completion since then counts 

63 if existing.cleared_at and finished <= existing.cleared_at: 

64 return NOT_YET 

65 existing.completed_at, existing.last_checked_at = finished, now 

66 existing.cleared_at = existing.cleared_by = None 

67 existing.save() 

68 else: 

69 TrainingCompletion.objects.create( 

70 member=member, moodle_course_id=course_id, completed_at=finished, last_checked_at=now 

71 ) 

72 machines = ", ".join(Equipment.objects.filter(moodle_course_id=course_id).values_list("name", flat=True)) 

73 notify( 

74 member, 

75 f"You've finished the online course for {machines}. A trainer can now sign you off.", 

76 kind="course_completed", 

77 key=f"moodle-course-{course_id}-{int(finished.timestamp())}", 

78 ) 

79 return COMPLETED

89def waiting_for_sign_off():

106def check_waiting(now=None, limit=None) -> Summary: 

107 """Check everyone waiting for sign-off. Stops at the first time Moodle can't be reached, so an outage costs one 

108 timeout rather than one per member.""" 

109 summary = Summary() 

110 for member, course_id in waiting_for_sign_off(): 

111 if limit is not None and summary.checked >= limit: 

112 break 

113 outcome = check(member, course_id, now) 

114 if outcome == UNAVAILABLE: 

115 summary.unavailable = True 

116 break 

117 summary.checked += 1 

118 summary.completed += outcome == COMPLETED 

119 return summary
