92def sign_off_problem(member, category: str, trained_at) -> str: 

93 """Why this member can't be signed off as trained on an induction category yet, as a sentence, or "" (05 §5.8). 

94 

95 Their email address must be confirmed (03 §3.1), and the online course finished where a machine in the category 

96 has one (06 §6.12). Used by the admin and by trainers on the machine's page. 

97 """ 

98 if not member.email_verified: 

99 return f"{member} hasn't confirmed their email address yet, so they can't be signed off." 

100 signing_off = Induction(member=member, category=category, trained_at=trained_at) 

101 for machine in Equipment.objects.filter(moodle_course_id__isnull=False, archived=False):

104 return ""

107def machine_refusal_reason(member, equipment: Equipment, *, for_offline_cache=False) -> str: 

108 """Why this member can't start a session on this machine, or "" if they can (05 §5.3). 

109 

110 for_offline_cache=True ignores credit and machine status: who a reader may remember (spec 09 §9.7). 

111 """ 

112 if not member.has_space_access: 

113 return "not_member" 

114 on_team = is_on_team(member, equipment) 

115 if equipment.status == Equipment.Status.OUT_OF_ORDER and not on_team and not for_offline_cache: 

116 return "out_of_order" 

117 if equipment.requires_induction: 

118 induction = Induction.objects.filter(member=member, category=equipment.induction_key).first() 

119 if induction is None or not induction.is_trained: 

120 return "not_trained" 

121 reason = _course_reason(member, equipment, induction) 

122 if reason: 

123 return reason 

124 if for_offline_cache: 

125 return "" 

126 credit_limit = SiteSettings.load().machine_credit_limit_pence 

127 if equipment.usage_cost_pence_per_hour and member_credit_pence(member) + credit_limit <= 0: 

128 return "insufficient_credit" 

129 return ""

137def _fee(equipment: Equipment, reason, removed, duration_seconds) -> int: 

138 if removed or reason in UNCHARGED_REASONS or not equipment.usage_cost_pence_per_hour: 

139 return 0 

140 seconds = max(duration_seconds, minimum_charge_seconds()) 

141 # Integer maths, rounding half up: rate is pence per hour. 

142 return (equipment.usage_cost_pence_per_hour * seconds + 1800) // 3600

145def fee_so_far_pence(session: UsageSession, now=None) -> int: 

146 """What an open session would cost if it ended now (at least the minimum charge). Spec 09 §9.5.""" 

147 now = now or timezone.now() 

148 seconds = max(0, int((now - session.started_at).total_seconds()))

168def start_session(equipment: Equipment, member, fob=None, *, close_existing: bool) -> UsageSession: 

169 """Start a session. Assumes machine_refusal_reason() said yes. 

170 

171 close_existing=True is the legacy (v1) behaviour: a new tap ends whatever session was open on the machine. 

172 v2 refuses a second member instead (decided 2026-11-07), so it passes False after checking. 

173 """ 

174 current = open_session(equipment) 

175 if current is not None and close_existing: 

176 end_session(current, ended_by=UsageSession.EndedBy.DEVICE) 

177 reason = UsageSession.Reason.NORMAL 

178 if equipment.status == Equipment.Status.OUT_OF_ORDER: 

179 reason = UsageSession.Reason.MAINTENANCE # only the team can get here (05 §5.4) 

180 now = timezone.now() 

181 return UsageSession.objects.create( 

182 equipment=equipment, member=member, fob=fob, started_at=now, last_heartbeat_at=now, reason=reason

192def end_session(session: UsageSession, *, when=None, ended_by: str) -> UsageSession: 

193 """End a session and charge its fee to the member's credit straight away (05 §5.4).""" 

194 session.ended_at = when or timezone.now() 

195 session.ended_by = ended_by 

196 session.removed = (session.ended_at - session.started_at) <= SHORT_SESSION 

197 session.fee_pence = session_fee_pence(session.equipment, session) 

198 session.save(update_fields=["ended_at", "ended_by", "removed", "fee_pence"]) 

199 services.settle_session(session) 

200 return session

203def running_seconds(equipment: Equipment, since=None, until=None) -> int: 

204 """How long the machine has run in a range (spec 09 §9.4). Worked out each time, so it can't drift.""" 

205 if equipment.meter_mode == Equipment.MeterMode.CURRENT:

208 readings = readings.filter(received_at__gte=since) 

209 if until: 209 ↛ 210line 209 didn't jump to line 210 because the condition on line 209 was never true

210 readings = readings.filter(received_at__lte=until) 

211 return readings.aggregate(total=Sum("active_seconds"))["total"] or 0 

212 now = timezone.now() 

213 total = 0 

214 sessions = equipment.sessions.filter(removed=False) 

215 if since: 

216 sessions = sessions.exclude(ended_at__lt=since) 

217 if until: 217 ↛ 218line 217 didn't jump to line 218 because the condition on line 217 was never true

218 sessions = sessions.filter(started_at__lte=until) 

219 for session in sessions: 

220 start = max(session.started_at, since) if since else session.started_at 

221 end = session.ended_at or now 

222 if until: 222 ↛ 223line 222 didn't jump to line 223 because the condition on line 222 was never true

223 end = min(end, until) 

224 total += max(0, int((end - start).total_seconds()))

251def auto_end_idle_sessions(now=None) -> int: 

252 """End sessions on current-draw machines whose motor stopped long ago (05 §5.4, forgotten tap-outs). 

253 

254 The session ends at the last moment the motor was running, not when we noticed. 

255 """ 

256 now = now or timezone.now() 

257 ended = 0 

258 machines = Equipment.objects.filter( 

259 meter_mode=Equipment.MeterMode.CURRENT, idle_auto_end_minutes__isnull=False, archived=False 

260 ) 

261 for equipment in machines: 

262 with transaction.atomic(): 

263 locked = Equipment.objects.select_for_update().get(pk=equipment.pk) 

264 session = open_session(locked) 

265 if session is None: 265 ↛ 266line 265 didn't jump to line 266 because the condition on line 265 was never true

266 continue 

267 last_running = ( 

268 locked.meter_readings.filter(active_seconds__gt=0, received_at__gt=session.started_at) 

269 .order_by("-received_at") 

270 .first() 

271 ) 

272 last_moment = last_running.received_at if last_running else session.started_at 

273 if now - last_moment >= timedelta(minutes=locked.idle_auto_end_minutes): 

274 end_session(session, when=last_moment, ended_by=UsageSession.EndedBy.IDLE) 

275 ended += 1 

276 return ended

279@transaction.atomic 

280def tidy_up(now=None) -> dict: 

281 """The hourly tidy-up (05 §5.6). Returns counts, for the job's log.""" 

282 now = now or timezone.now() 

283 counts = {"closed": 0, "merged": 0, "idle": 0, "reminders": 0} 

284 counts["idle"] = auto_end_idle_sessions(now) 

285 counts["reminders"] = check_service_reminders() 

286 

287 # 1. Close open sessions that have gone quiet, at their last heartbeat (or a minute after starting). 

288 for session in UsageSession.objects.filter(ended_at__isnull=True).select_related("equipment"): 

289 last_seen = session.last_heartbeat_at or session.started_at 

290 if now - last_seen > STALE_AFTER: 

291 end_at = session.last_heartbeat_at or (session.started_at + timedelta(minutes=1)) 

292 end_session(session, when=end_at, ended_by=UsageSession.EndedBy.TIDY_UP) 

293 counts["closed"] += 1 

294 

295 # 2. Merge back-to-back sessions by the same member on the same machine, with the same reason, so a member who 

296 # taps out and straight back in pays one minimum charge, not two. Only recent sessions, and never ones the 

297 # legacy system settled. The fees are put right with a correcting ledger row (settle_session). 

298 finished = ( 

299 UsageSession.objects.filter(ended_at__isnull=False, ended_at__gte=now - MERGE_WINDOW) 

300 .exclude(legacy_id__isnull=False, charged_at__isnull=False) 

301 .select_related("equipment") 

302 .order_by("equipment_id", "member_id", "reason", "started_at") 

303 ) 

304 previous = None 

305 for session in finished: 

306 same_run = ( 

307 previous is not None 

308 and (previous.equipment_id, previous.member_id, previous.reason) 

309 == (session.equipment_id, session.member_id, session.reason) 

310 and session.started_at >= previous.ended_at 

311 and session.started_at - previous.ended_at <= MERGE_GAP 

312 ) 

313 if same_run: 

314 previous.ended_at = session.ended_at 

315 previous.ended_by = session.ended_by 

316 previous.motor_on_seconds += session.motor_on_seconds 

317 previous.removed = (previous.ended_at - previous.started_at) <= SHORT_SESSION 

318 previous.fee_pence = session_fee_pence(previous.equipment, previous) 

319 previous.save(update_fields=["ended_at", "ended_by", "motor_on_seconds", "removed", "fee_pence"]) 

320 # Keep the merged session's charges and meter readings with the session that carries on. 

321 session.payments.update(usage_session=previous) 

322 MeterReading.objects.filter(session=session).update(session=previous) 

323 session.delete() 

324 services.settle_session(previous) 

325 counts["merged"] += 1 

326 else: 

327 previous = session 

328 return counts
