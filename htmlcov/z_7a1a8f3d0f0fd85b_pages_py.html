1"""

55def training_state(member, equipment: Equipment) -> TrainingState: 

56 if not equipment.requires_induction: 

57 return TrainingState("none") 

58 induction = Induction.objects.filter(member=member, category=equipment.induction_key).first() 

59 if induction is None or not induction.paid: 

60 step = "unpaid" 

61 else: 

62 step = "trained" if induction.is_trained else "waiting" 

63 state = TrainingState(step, induction) 

64 if equipment.moodle_course_id: 

65 state.course_done = TrainingCompletion.objects.filter( 

66 member=member, moodle_course_id=equipment.moodle_course_id, cleared_at__isnull=True 

67 ).exists() 

68 if not state.course_done: 

69 check = induction or Induction(member=member, category=equipment.induction_key) 

70 missing = rules._course_reason(member, equipment, check) != "" 

71 state.course_kept = step == "trained" and not missing 

72 state.course_needed = missing 

73 return state

96def usage_minutes(equipment: Equipment) -> dict: 

97 """Total time used, split as legacy did (05 §5.7): charged use, training, testing and maintenance.""" 

98 finished = equipment.sessions.filter(ended_at__isnull=False, removed=False) 

99 totals = {"used": 0, "training": 0, "testing": 0, "maintenance": 0} 

100 for reason, started, ended in finished.values_list("reason", "started_at", "ended_at"): 

101 key = reason or "used" 

102 totals[key] = totals.get(key, 0) + int((ended - started).total_seconds()) 

103 return {key: seconds // 60 for key, seconds in totals.items()}
