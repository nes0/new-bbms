80def signature_ok(body: bytes, signature: str) -> bool: 

81 secret = settings.GOCARDLESS_WEBHOOK_SECRET 

82 if not secret or not signature: 

83 return False 

84 expected = hmac.new(secret.encode(), body, hashlib.sha256).hexdigest() 

85 return hmac.compare_digest(expected, signature) # constant time
