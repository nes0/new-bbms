143class _FormValues(HTMLParser): 

144 """The values a browser would submit from a page's form with the given id.""" 

145 

146 def __init__(self, form_id): 

147 super().__init__() 

148 self.form_id, self.inside, self.values = form_id, False, [] 

149 self._select = self._textarea = None 

150 

151 def handle_starttag(self, tag, attrs): 

152 attrs = dict(attrs) 

153 if tag == "form": 

154 self.inside = attrs.get("id") == self.form_id 

155 if not self.inside: 

156 return 

157 if tag == "input" and attrs.get("name") and attrs.get("type") not in ("submit", "button", "file"): 

158 if attrs.get("type") in ("checkbox", "radio"): 

159 if "checked" in attrs: 

160 self.values.append((attrs["name"], attrs.get("value") or "on")) 

161 else: 

162 self.values.append((attrs["name"], attrs.get("value") or "")) 

163 elif tag == "select": 

164 self._select = attrs.get("name") 

165 elif tag == "option" and self._select and "selected" in attrs: 

166 self.values.append((self._select, attrs.get("value", ""))) 

167 elif tag == "textarea": 

168 self._textarea = attrs.get("name") 

169 self.values.append((self._textarea, "")) 

170 

171 def handle_data(self, data): 

172 if self._textarea: 

173 name, value = self.values.pop() 

174 self.values.append((name, value + data)) 

175 

176 def handle_endtag(self, tag): 

177 if tag == "form": 

178 self.inside = False 

179 elif tag == "select": 

180 self._select = None 

181 elif tag == "textarea": 

182 self._textarea = None

192 def save_admin_page(self, **changes): 

193 """Submit the member's admin page as it is, with `changes` (True/False tick or untick a box).""" 

194 parser = _FormValues("member_form") 

195 parser.feed(self.client.get(self.url).content.decode()) 

196 data = {} 

197 for name, value in parser.values: 

198 data.setdefault(name, []).append(value) 

199 for name, value in changes.items(): 

200 if value is True: 

201 data[name] = ["on"] 

202 elif value is False: 202 ↛ 203line 202 didn't jump to line 203 because the condition on line 202 was never true

203 data.pop(name, None) 

204 else: 

205 data[name] = [value] 

206 return self.client.post(self.url, data)
