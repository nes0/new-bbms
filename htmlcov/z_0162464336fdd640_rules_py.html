292def first_rent(member, monthly_pence: int, today: date | None = None) -> tuple[int, date]: 

293 """The first payment for a locker rented today: the rent for the days until the member's next payment day, and 

294 that day (❓ 06 §6.2, the suggested rule). Renting on the payment day itself pays for a whole month. 

295 """ 

296 from apps.billing.membership import add_month, payment_day_for 

297 

298 today = today or timezone.localdate() 

299 day = member.payment_day or payment_day_for(today) 

300 this_month = today.replace(day=day) 

301 if this_month > today: 

302 # Last month's payment day (payment days are 1-28, so it exists in every month). 

303 month_before = this_month.replace(day=1) - timedelta(days=1) 

304 next_day, previous = this_month, month_before.replace(day=day) 

305 else: 

306 next_day, previous = add_month(this_month), this_month 

307 days, period = (next_day - today).days, (next_day - previous).days 

308 return (monthly_pence * days + period // 2) // period, next_day
