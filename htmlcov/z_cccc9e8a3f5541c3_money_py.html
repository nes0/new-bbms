1""" 

2What the space took and spent, by heading (spec 04 §4.9), and what each machine has made (requested by the 

3trustees, 2026-11-08). 

4 

5**Income** is counted when it's paid for something, by what it was for, whatever paid it: a laser fee taken from 

6credit is income when the laser is used, so a credit top-up isn't income too (it's money held for the member, shown 

7separately). Payments that are paid count (and legacy's "withdrawn"); pending ones (a Direct Debit on its way) are 

8shown separately. Corrections are negative rows under the heading they correct. Dates are when BBMS recorded the 

9payment. 

10 

11**Expenditure** is what BBMS knows about: expenses members claimed back, counted when they were approved. The 

12space's other spending (rent, bills) isn't in BBMS. 

13 

14**Machines**: induction fees + usage fees − approved expenses claimed for the machine. Machines that share an 

15induction (several lasers on one induction, say) are one row, because the induction fee is for all of them. 

16"""

219def received(start: date, end: date) -> dict: 

220 """What reached the bank from each source, by the day it arrived, as BBMS knows it and as the bank statements say, 

221 so the two can be compared line by line. Payouts are dated by when they arrive at the bank, not when members 

222 paid.""" 

223 from .models import BankLine, Payout 

224 

225 payouts = Payout.objects.filter(arrival_date__gte=start, arrival_date__lte=end).exclude( 

226 status__in=("bounced", "failed", "canceled") 

227 ) 

228 lines = BankLine.objects.filter(kept=True, date__gte=start, date__lte=end) 

229 since, until = _range(start, end) 

230 paid = Payment.objects.filter(status__in=COUNTED, paid_at__gte=since, paid_at__lt=until) 

231 rows = [] 

232 for key, label, kind, known in RECEIVED: 

233 bank = lines.filter(kind=kind).aggregate(t=Sum("amount_pence"))["t"] or 0 if kind else None 

234 if known == "payouts": 

235 ours = payouts.filter(provider=key).aggregate(t=Sum("amount_pence"))["t"] or 0 

236 elif known == "standing-order": 

237 ours = paid.filter(source=Payment.Source.STANDING_ORDER).aggregate(t=Sum("amount_pence"))["t"] or 0 

238 elif known == "cash": 

239 ours = ( 

240 paid.filter(source=Payment.Source.CASH, amount_pence__gt=0).aggregate(t=Sum("amount_pence"))["t"] or 0 

241 ) 

242 else: 

243 ours = None 

244 difference = bank - ours if bank is not None and ours is not None else None 

245 rows.append({"key": key, "label": label, "bbms": ours, "bank": bank, "difference": difference}) 

246 has_statement = lines.exists() 

247 return { 

248 "rows": rows, 

249 "bank_total": sum(row["bank"] or 0 for row in rows), 

250 "has_statement": has_statement, 

251 "unmatched_lines": lines.filter(outcome=BankLine.Outcome.UNMATCHED).order_by("date"), 

252 "payout_lines_alone": lines.filter( 

253 kind__in=(BankLine.Kind.GOCARDLESS, BankLine.Kind.STRIPE), payout_match__isnull=True 

254 ).order_by("date"), 

255 # Only worth listing once a statement for the period has been imported. 

256 "payouts_not_on_statement": payouts.filter(bank_line__isnull=True, status="paid").order_by("arrival_date") 

257 if has_statement 

258 else Payout.objects.none(), 

259 }
