1""" 

2Claiming, approving and declining expenses (spec 06 §6.4). 

3 

4- **Claim**: members (anyone who sees the member directory) claim for themselves only, with a receipt (JPEG, PNG or 

5 PDF). Photos of receipts are re-saved, which drops the camera's metadata (e.g. where it was taken). Everyone who 

6 can approve expenses is told (in BBMS and by email). 

7- **Approve** (the "approve expenses" permission; admins have it): pays the amount into the member's credit, once. 

8 Nobody approves their own claim. 

9- **Decline**: with a reason, which the member is sent. 

10- A member can **withdraw** a claim that's still waiting; it's deleted with its receipt. 

11- **Who sees what**: members see their own claims; those who can approve, and those with "see all expenses" (the 

12 finance role), see everyone's. 

13"""

165def _decide(expense_id, by, status, *, reason="") -> Expense: 

166 if not can_approve(by): 

167 raise CantDecide("You can't approve or decline expenses.") 

168 with transaction.atomic(): 

169 expense = Expense.objects.select_for_update().select_related("member").get(pk=expense_id) 

170 if expense.status != Expense.Status.WAITING: 

171 raise CantDecide(f"This expense has already been {expense.get_status_display().lower()}.") 

172 if expense.member_id == by.pk: 

173 raise CantDecide("Someone else must decide on your own expense.") 

174 if status == Expense.Status.APPROVED: 

175 expense.payment = services.record_payment( 

176 expense.member, 

177 Payment.Reason.BALANCE, 

178 Payment.Source.REIMBURSEMENT, 

179 expense.amount_pence, 

180 reference=str(expense.pk), 

181 created_by=by, 

182 ) 

183 expense.status, expense.decided_by, expense.decided_at = status, by, timezone.now() 

184 expense.decline_reason = reason 

185 expense.save() 

186 logger.info("Expense %s %s by member %s", expense.pk, status, by.pk)
